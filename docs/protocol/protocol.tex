% Typesetting for the preregistration. The OPPOSITE brief from proposal/tighten.tex:
% the 2-pager fights for vertical space against a hard page limit, and this document
% has no page limit and one job, which is to be read carefully by someone deciding
% whether the study is registered. So it gets air, a table of contents, and rules
% around tables rather than squeezed leading.
\usepackage{parskip}

% Helvetica Neue HAS NO U+2192. xelatex drops the glyph and warns, it does not fail,
% so the first render of this document silently turned every rubric rule of the form
%     "single-instrument validation in a domain -> OUT"
% into "single-instrument validation in a domain  OUT": a rule with its verb deleted,
% in the appendix whose entire job is to say what maps to what. A missing glyph in a
% coding manual is not a typographic complaint, it is a changed instruction, so the
% arrows are mapped to glyphs the font actually has.
\usepackage{newunicodechar}
\newunicodechar{→}{\ensuremath{\rightarrow}}
\newunicodechar{←}{\ensuremath{\leftarrow}}
\newunicodechar{≥}{\ensuremath{\geq}}
\newunicodechar{≤}{\ensuremath{\leq}}
\newunicodechar{≈}{\ensuremath{\approx}}
\newunicodechar{≠}{\ensuremath{\neq}}
\newunicodechar{×}{\ensuremath{\times}}
\newunicodechar{∩}{\ensuremath{\cap}}
\newunicodechar{∪}{\ensuremath{\cup}}
\newunicodechar{∈}{\ensuremath{\in}}
\newunicodechar{κ}{\ensuremath{\kappa}}
\newunicodechar{α}{\ensuremath{\alpha}}
\newunicodechar{β}{\ensuremath{\beta}}
\newunicodechar{π}{\ensuremath{\pi}}
\newunicodechar{√}{\ensuremath{\surd}}
\newunicodechar{∞}{\ensuremath{\infty}}

\usepackage{array}
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{5pt}

\usepackage{enumitem}
\setlist[enumerate]{topsep=3pt, itemsep=2pt, parsep=0pt, leftmargin=*}
\setlist[itemize]{topsep=3pt, itemsep=2pt, parsep=0pt, leftmargin=*}

% A protocol is quoted from constantly (the rubric's worked examples, the estimand).
% Set blockquotes apart so a quoted rule is visibly a rule and not narration.
\usepackage{etoolbox}
\usepackage{xcolor}
\definecolor{rulegray}{HTML}{888888}
\AtBeginEnvironment{quote}{\vspace{2pt}\itshape\color{black!80}}
\AtEndEnvironment{quote}{\vspace{2pt}}

% Running head: the version of the document is the single most important thing to
% see on any page a reviewer prints, because "which rubric was this screened under"
% is the question the whole appendix structure exists to answer.
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\footnotesize\color{rulegray} MétaCan study protocol}
\fancyhead[R]{\footnotesize\color{rulegray} preregistration draft}
\fancyfoot[C]{\footnotesize\thepage}
\renewcommand{\headrulewidth}{0.4pt}
