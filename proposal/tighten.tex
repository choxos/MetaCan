% Vertical economy for a hard two-page limit. Body text stays at 10pt and the
% measure stays readable; what shrinks is the whitespace around headings, lists
% and table rows, which is where a rendered Markdown document wastes its page.
\usepackage{titlesec}
\titlespacing*{\section}{0pt}{5pt plus 1pt minus 1pt}{2pt}
\titleformat*{\section}{\large\bfseries}

\setlength{\parskip}{1pt plus 1pt minus 0.5pt}
\setlength{\parindent}{0pt}

% Tables: trim row padding and let the two-column findings table use the full measure.
\usepackage{array}
\renewcommand{\arraystretch}{1.05}
\setlength{\tabcolsep}{4pt}

% Lists: kill the extra leading above/below and between items.
\usepackage{enumitem}
\setlist[enumerate]{topsep=2pt, itemsep=1pt, parsep=0pt, partopsep=0pt, leftmargin=*}
\setlist[itemize]{topsep=2pt, itemsep=1pt, parsep=0pt, partopsep=0pt, leftmargin=*}

% The title block is not worth a third of a page.
\usepackage{titling}
\setlength{\droptitle}{-2.6cm}
\pretitle{\begin{center}\large\bfseries}
\posttitle{\end{center}\vspace{-0.6em}}
\predate{}\postdate{}\date{}

% Blockquotes: pandoc wraps them in `quote`, which adds generous vertical space.
\usepackage{etoolbox}
\AtBeginEnvironment{quote}{\vspace{-4pt}\small}
\AtEndEnvironment{quote}{\vspace{-4pt}}
\setlength{\topsep}{2pt}
\setlength{\partopsep}{0pt}

% 10pt body is kept; what tightens is leading, not type size. 0.97 is
% imperceptible at reading distance and buys back the ~6 lines the pilot's
% results added. The alternative was cutting a finding, which is the wrong trade.
\linespread{0.95}
